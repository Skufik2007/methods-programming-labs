```python
from datetime import date, time, timedelta

import pytest

from minutes import available_minutes

# 2024-01-01 — понедельник (isoweekday == 1)
MON = date(2024, 1, 1)
WORKDAYS = frozenset({1, 2, 3, 4, 5})
ALL_DAYS = frozenset(range(1, 8))
T9 = time(9, 0)
T17 = time(17, 0)
SHIFT = 8 * 60


def reference(work_days, start, end, d_from, d_to):
    shift = (end.hour * 60 + end.minute) - (start.hour * 60 + start.minute)
    total = 0
    d = d_from
    while d <= d_to:
        if d.isoweekday() in work_days:
            total += shift
        d += timedelta(days=1)
    return total


# ---------- базовые случаи ----------


def test_single_working_day():
    assert available_minutes(WORKDAYS, T9, T17, MON, MON) == SHIFT


def test_single_non_working_day():
    saturday = date(2024, 1, 6)
    assert saturday.isoweekday() == 6
    assert available_minutes(WORKDAYS, T9, T17, saturday, saturday) == 0


def test_single_day_each_weekday_number():
    # каждый день недели проверяется отдельно: работает только он
    for offset in range(7):
        d = MON + timedelta(days=offset)
        wd = d.isoweekday()
        assert wd == offset + 1
        assert available_minutes(frozenset({wd}), T9, T17, d, d) == SHIFT
        other = frozenset(set(range(1, 8)) - {wd})
        assert available_minutes(other, T9, T17, d, d) == 0


def test_sunday_is_seven():
    sunday = date(2024, 1, 7)
    assert sunday.isoweekday() == 7
    assert available_minutes(frozenset({7}), T9, T17, sunday, sunday) == SHIFT
    assert available_minutes(frozenset({7}), T9, T17, MON, sunday) == SHIFT


def test_full_week_mon_to_sun():
    assert available_minutes(WORKDAYS, T9, T17, MON, date(2024, 1, 7)) == 5 * SHIFT


def test_two_full_weeks():
    assert available_minutes(WORKDAYS, T9, T17, MON, date(2024, 1, 14)) == 10 * SHIFT


def test_many_full_weeks_different_start_day():
    # 4 полные недели начиная со среды
    start = date(2024, 1, 3)
    end = start + timedelta(days=28 - 1)
    assert available_minutes(frozenset({1, 3, 6}), T9, T17, start, end) == 12 * SHIFT


def test_period_includes_both_boundaries():
    # пн..пт включительно -> 5 дней, а не 4 и не 6
    assert available_minutes(WORKDAYS, T9, T17, MON, date(2024, 1, 5)) == 5 * SHIFT
    # граничный выходной не должен учитываться, соседний рабочий — должен
    assert available_minutes(frozenset({1}), T9, T17, MON, MON + timedelta(days=1)) == SHIFT
    assert available_minutes(frozenset({2}), T9, T17, MON, MON + timedelta(days=1)) == SHIFT


# ---------- неполные недели ----------


def test_partial_week_start_midweek():
    # ср..вс (5 дней), работают пн-пт -> ср, чт, пт = 3
    assert available_minutes(WORKDAYS, T9, T17, date(2024, 1, 3), date(2024, 1, 7)) == 3 * SHIFT


def test_partial_week_wraps_over_sunday():
    # сб 2024-01-06 .. вт 2024-01-09: сб, вс, пн, вт
    assert available_minutes(WORKDAYS, T9, T17, date(2024, 1, 6), date(2024, 1, 9)) == 2 * SHIFT


def test_remainder_wraps_with_full_week():
    # 10 дней: пт 2024-01-05 .. пн 2024-01-14 (пт сб вс пн вт ср чт пт сб вс -> 10 дней)
    d_from, d_to = date(2024, 1, 5), date(2024, 1, 14)
    assert (d_to - d_from).days + 1 == 10
    assert available_minutes(WORKDAYS, T9, T17, d_from, d_to) == reference(WORKDAYS, T9, T17, d_from, d_to)
    assert available_minutes(WORKDAYS, T9, T17, d_from, d_to) == 6 * SHIFT


def test_rest_days_only_weekend():
    # 8 дней: сб .. сб; работают пн-пт -> 5 рабочих в полной неделе + 0 в остатке
    d_from, d_to = date(2024, 1, 6), date(2024, 1, 13)
    assert available_minutes(WORKDAYS, T9, T17, d_from, d_to) == 5 * SHIFT


def test_rest_length_six_days():
    # 6 дней: пн..сб
    assert available_minutes(WORKDAYS, T9, T17, MON, date(2024, 1, 6)) == 5 * SHIFT
    # 6 дней: вт..пн -> вт ср чт пт сб вс
    assert available_minutes(WORKDAYS, T9, T17, date(2024, 1, 2), date(2024, 1, 7)) == 4 * SHIFT


def test_seven_days_exactly_one_week_any_start():
    for offset in range(7):
        d_from = MON + timedelta(days=offset)
        d_to = d_from + timedelta(days=6)
        assert available_minutes(frozenset({1, 4}), T9, T17, d_from, d_to) == 2 * SHIFT


def test_eight_days_one_week_plus_one():
    # пн..пн: работает только пн -> 2 дня
    assert available_minutes(frozenset({1}), T9, T17, MON, date(2024, 1, 8)) == 2 * SHIFT
    # работает только вт -> 1 день
    assert available_minutes(frozenset({2}), T9, T17, MON, date(2024, 1, 8)) == SHIFT


# ---------- наборы рабочих дней ----------


def test_empty_work_days():
    assert available_minutes(frozenset(), T9, T17, MON, date(2024, 3, 31)) == 0


def test_all_days_work():
    d_to = date(2024, 1, 10)  # 10 дней
    assert available_minutes(ALL_DAYS, T9, T17, MON, d_to) == 10 * SHIFT


def test_non_contiguous_work_days():
    days = frozenset({1, 3, 5, 7})
    # пн..вс: 4 дня; следующий пн-ср: пн, ср -> 2
    assert available_minutes(days, T9, T17, MON, date(2024, 1, 7)) == 4 * SHIFT
    assert available_minutes(days, T9, T17, MON, date(2024, 1, 10)) == 6 * SHIFT


def test_only_weekend_work_days():
    days = frozenset({6, 7})
    assert available_minutes(days, T9, T17, MON, date(2024, 1, 14)) == 4 * SHIFT


def test_weekend_only_period_with_weekday_schedule():
    assert available_minutes(WORKDAYS, T9, T17, date(2024, 1, 6), date(2024, 1, 7)) == 0


# ---------- длина смены ----------


def test_shift_with_minutes():
    assert available_minutes(frozenset({1}), time(8, 30), time(12, 15), MON, MON) == 225


def test_shift_with_minutes_multiple_days():
    assert available_minutes(WORKDAYS, time(8, 45), time(17, 10), MON, date(2024, 1, 5)) == 5 * (8 * 60 + 25)


def test_zero_length_shift():
    assert available_minutes(WORKDAYS, T9, T9, MON, date(2024, 1, 31)) == 0


def test_one_minute_shift():
    assert available_minutes(ALL_DAYS, time(9, 0), time(9, 1), MON, date(2024, 1, 7)) == 7


def test_full_day_shift():
    assert available_minutes(frozenset({1}), time(0, 0), time(23, 59), MON, MON) == 23 * 60 + 59


def test_seconds_in_times_do_not_add_extra_minutes():
    # 09:00:00 .. 09:00:59 -> неполная минута не считается
    assert available_minutes(frozenset({1}), time(9, 0, 0), time(9, 0, 59), MON, MON) == 0
    assert available_minutes(frozenset({1}), time(9, 0, 0), time(9, 1, 30), MON, MON) == 1


def test_result_is_int():
    result = available_minutes(WORKDAYS, T9, T17, MON, date(2024, 1, 5))
    assert isinstance(result, int)


# ---------- календарные особенности ----------


def test_across_month_boundary():
    d_from, d_to = date(2024, 1, 29), date(2024, 2, 4)  # пн..вс
    assert available_minutes(WORKDAYS, T9, T17, d_from, d_to) == 5 * SHIFT


def test_across_year_boundary():
    d_from, d_to = date(2023, 12, 28), date(2024, 1, 3)  # чт..ср, 7 дней
    assert available_minutes(WORKDAYS, T9, T17, d_from, d_to) == reference(WORKDAYS, T9, T17, d_from, d_to)
    assert available_minutes(WORKDAYS, T9, T17, d_from, d_to) == 5 * SHIFT


def test_leap_day():
    d = date(2024, 2, 29)  # четверг
    assert d.isoweekday() == 4
    assert available_minutes(WORKDAYS, T9, T17, d, d) == SHIFT
    assert available_minutes(WORKDAYS, T9, T17, date(2024, 2, 28), date(2024, 3, 1)) == 3 * SHIFT


def test_long_period_full_year():
    d_from, d_to = date(2024, 1, 1), date(2024, 12, 31)  # 366 дней = 52 нед + 2 дня (пн, вт)
    expected = reference(WORKDAYS, T9, T17, d_from, d_to)
    assert expected == 262 * SHIFT
    assert available_minutes(WORKDAYS, T9, T17, d_from, d_to) == expected


# ---------- неверный период ----------


def test_reversed_period_raises():
    with pytest.raises(ValueError):
        available_minutes(WORKDAYS, T9, T17, date(2024, 1, 2), MON)


def test_reversed_period_by_one_day_raises():
    with pytest.raises(ValueError):
        available_minutes(ALL_DAYS, T9, T17, date(2024, 1, 10), date(2024, 1, 9))


def test_reversed_period_raises_even_with_empty_work_days():
    with pytest.raises(ValueError):
        available_minutes(frozenset(), T9, T17, date(2024, 5, 2), date(2024, 5, 1))


def test_equal_dates_do_not_raise():
    assert available_minutes(WORKDAYS, T9, T17, MON, MON) == SHIFT


# ---------- сверка с эталоном перебором ----------


@pytest.mark.parametrize(
    "work_days",
    [
        frozenset({1, 2, 3, 4, 5}),
        frozenset({1}),
        frozenset({7}),
        frozenset({2, 4, 6}),
        frozenset({1, 7}),
        frozenset(range(1, 8)),
        frozenset(),
    ],
)
@pytest.mark.parametrize("start_offset", range(7))
@pytest.mark.parametrize("length", [1, 2, 3, 6, 7, 8, 9, 13, 14, 15, 20, 21, 22, 30, 31, 100])
def test_matches_bruteforce(work_days, start_offset, length):
    d_from = MON + timedelta(days=start_offset)
    d_to = d_from + timedelta(days=length - 1)
    start, end = time(8, 15), time(16, 40)
    assert available_minutes(work_days, start, end, d_from, d_to) == reference(work_days, start, end, d_from, d_to)


def test_additivity_over_split_period():
    d_from, d_to = date(2024, 3, 5), date(2024, 6, 20)
    mid = date(2024, 4, 17)
    whole = available_minutes(WORKDAYS, T9, T17, d_from, d_to)
    parts = available_minutes(WORKDAYS, T9, T17, d_from, mid) + available_minutes(
        WORKDAYS, T9, T17, mid + timedelta(days=1), d_to
    )
    assert whole == parts
```
