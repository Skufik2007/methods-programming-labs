```python
import re

_SNILS_RE = re.compile(r"[0-9]{3}-[0-9]{3}-[0-9]{3} [0-9]{2}|[0-9]{11}")


def validate_snils(value: str) -> bool:
    if not isinstance(value, str):
        return False
    m = _SNILS_RE.fullmatch(value.strip(" "))
    if m is None:
        return False
    digits = [int(c) for c in re.sub(r"[^0-9]", "", m.group())]
    number = int("".join(map(str, digits[:9])))
    if number <= 1_001_998:  # 001-001-998
        return True
    total = sum(d * (9 - i) for i, d in enumerate(digits[:9]))
    if total < 100:
        control = total
    else:
        control = total % 101
        if control == 100:
            control = 0
    return control == digits[9] * 10 + digits[10]
```

Функция проверяет формат ("112-233-445 95" или "11223344595", пробелы по краям допустимы), для номеров больше 001-001-998 сверяет контрольное число по алгоритму ПФР, а меньшие или равные считает корректными при верном формате.
