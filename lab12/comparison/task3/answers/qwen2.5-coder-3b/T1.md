```python
from typing import *
from collections import *


def validate_snils(value: str) -> bool:
    # Удаление пробелов и проверка формата
    value = value.replace(" ", "")
    if len(value) != 11 or not value.isdigit():
        return False

    # Проверка контрольного числа
    if int(value[-2:]) != 00:
        total = sum(int(digit) * (10 - index) for index, digit in enumerate(value[:-2], start=1))
        if total < 100:
            control_number = total
        elif total == 100 or total == 101:
            control_number = 00
        else:
            control_number = total % 101
            if control_number == 100:
                control_number = 00

        return control_number == int(value[-2:])

    return True
```
