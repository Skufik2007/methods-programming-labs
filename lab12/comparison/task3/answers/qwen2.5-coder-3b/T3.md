import pytest
from datetime import date, datetime, time
from minutes import available_minutes


def test_available_minutes_valid():
    work_days = frozenset([1, 2, 3, 4, 5])  # Понедельник - пятница
    work_start = time(9, 0)
    work_end = time(17, 0)
    date_from = date(2023, 10, 1)
    date_to = date(2023, 10, 5)
    expected = 4 * 8 * 60  # 4 дня * 8 часов * 60 минут
    assert available_minutes(work_days, work_start, work_end, date_from, date_to) == expected
