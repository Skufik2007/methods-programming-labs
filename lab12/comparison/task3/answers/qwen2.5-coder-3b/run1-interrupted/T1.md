from typing import *
from collections import *


def validate_snils(value: str) -> bool:
    # Удаляем пробелы и проверяем формат
    value = value.replace(" ", "")
    if len(value) != 11 or not value.isdigit():
        return False

    # Проверяем контрольное число
    if int(value[-2:]) != 00 and int(value[-2:]) != 100:
        total = sum((i + 1) * int(value[i]) for i in range(8))
        if total < 100:
            control_number = total
        elif total == 100 or total == 101:
            control_number = 00
        else:
            control_number = total % 101
            if control_number == 100:
                control_number = 00
        if int(value[-2:]) != control_number:
            return False

    # Проверяем диапазон
    if int(value) <= 101001998:
        return False

    return True
